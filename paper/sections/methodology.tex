\section{Methodology and Theoretical Formulation}

To resolve specimen-level data leakage while addressing taxon-specific structural heterogeneity, we formulate a comprehensive partitioning framework. First, we define the mechanisms of Same-Specimen-Picture Bias (SSPB). Next, we introduce a streamlined suite of representative splitting protocols spanning distinct algorithmic paradigms. Then, we construct the "Three Pillars Framework" comprising 16 quantitative metrics. Finally, we detail the mathematical formulation of our proposed Multi-Objective Simulated Annealing Meta-Selector ($\text{CEGS-Split}$).

\subsection{Problem Formulation and Specimen Leakage Risk (SLR)}
Let $\mathcal{D} = \{(x_i, y_i, g_i)\}_{i=1}^N$ represent a wood dataset of $N$ macro images, where $x_i \in \mathcal{X}$ is an RGB image observation, $y_i \in \{1, \dots, C\}$ is the biological species label ($C=18$), and $g_i \in \mathcal{G}$ denotes the physical specimen block (subfolder identifier).

A data partition $\pi: \mathcal{D} \to \{\text{Train}, \text{Val}, \text{Test}\}$ maps each sample $x_i$ to a split. Under naive random splitting, split assignment is independent of specimen origin $g_i$, inducing severe Same-Specimen-Picture Bias (SSPB). We quantify this vulnerability via the **Specimen Leakage Risk (SLR)**:
\begin{equation}
\mathrm{SLR}(\pi) = \frac{|\{g \in \mathcal{G} \mid \exists x, x' \in g \text{ s.t. } \pi(x) \neq \pi(x')\}|}{|\mathcal{G}|} \times 100\%,
\end{equation}
where $\mathrm{SLR} = 0.0\%$ indicates absolute physical specimen isolation, and $\mathrm{SLR} > 0.0\%$ measures the percentage of leaked physical specimens.

\begin{tcolorbox}[colback=lightbg,colframe=primaryblue,title=\textbf{SCDP Core Data Governance Axioms}]
\begin{enumerate}[leftmargin=*]
\item \textbf{100\% Subfolder Integrity}: All image observations $x_i, x_j$ originating from the same physical specimen $g \in \mathcal{G}$ MUST belong to the exact same split partition:
\begin{equation}
\forall g \in \mathcal{G}, \quad \pi(x_i) = \pi(x_j) \quad \forall x_i, x_j \in g.
\label{eq:subfolder_integrity}
\end{equation}
\item \textbf{100\% Class Coverage Rate (CCR)}: Every target species $c \in \{1, \dots, C\}$ MUST be present in Train, Validation, AND Test splits:
\begin{equation}
\mathrm{CCR}(\pi) = \frac{|\{c \in \{1,\dots,C\} \mid c \text{ present in Train, Val, and Test}\}|}{C} \times 100\% = 100.0\%.
\end{equation}
\end{enumerate}
\end{tcolorbox}

\subsection{Categorization of Evaluated Splitting Protocols}
To benchmark dataset partitioning paradigms without redundancy, we evaluate representative splitting protocols categorized into four distinct analytical blocks:

\begin{enumerate}[label=\textbf{Category \Roman*:}, leftmargin=*]
\item \textbf{Naive Image-Level Baseline (\textsf{Naive Random Image Split})}: Performs independent uniform random sampling over image files according to target ratios $r = (0.60, 0.20, 0.20)$. Serves as the maximum-leakage baseline ($\mathrm{SLR} = 100\%$).

\item \textbf{Single Splitting Protocols (Single Paradigm Imposed Globally)}:
\begin{itemize}
\item \textbf{\textsf{Stratified Group Split}}: Extends multi-objective group allocation to isolate physical specimen groups $g_i$ while maintaining per-class target split ratios.
\item \textbf{\textsf{Hierarchical Ward Partitioning}}: Computes specimen feature centroids $h_g = \frac{1}{|g|} \sum_{x \in g} \phi(x)$ using EfficientNetV2 embeddings. Hierarchical agglomerative clustering with Ward's minimum variance linkage is performed on $\{h_g\}$, assigning intact sub-trees to splits to minimize inter-cluster linkage distance.
\item \textbf{\textsf{Adversarial Density Validation}}: Trains an auxiliary binary MLP discriminator $D_\psi: \mathbb{R}^d \to [0, 1]$ to distinguish candidate split assignments. Specimens assigned high discriminator certainty $|D_\psi(h_g) - 0.5|$ are identified as distributional outliers and allocated to the Test set to enforce evaluation difficulty.
\item \textbf{\textsf{DataSAIL Specimen-Level ILP}}: Formulates specimen partitioning as a constrained 2-node / multi-node Graph Cut problem solved via Integer Linear Programming (ILP)~\cite{kalina2023}. Minimizes specimen centroid inter-split similarity:
\begin{equation}
\min_{\pi} \sum_{g \neq g'} \mathbb{1}[\pi(g) \neq \pi(g')] \cdot \cos(h_g, h_{g'}) \cdot |g| \cdot |g'| \quad \text{s.t. } \left| \frac{|W_s|}{W_{\text{total}}} - r_s \right| \le \epsilon.
\end{equation}
\end{itemize}

\item \textbf{Combinatorial Selector: Single-Objective DataSAIL Loss (\textsf{Single-Objective Classwise Selector})}: Evaluates 11 candidate solvers per taxon to independently minimize per-class DataSAIL loss $L_c(m_c)$.

\item \textbf{Combinatorial Selector: Multi-Objective Optimization (\textsf{Multi-Objective SA Meta-Selector}, Proposed)}: A combinatorial optimization engine operating over an $11^{18}$ search space using Simulated Annealing to discover global Pareto-optimal class-wise splitting configurations evaluated against a multi-objective fitness function.
\end{enumerate}

\subsection{Multi-Objective Simulated Annealing Meta-Selector ($k^N$ Search Space)}

\subsubsection{Taxon Heterogeneity Motivation (Why Class-wise Selection?)}
Wood taxa exhibit contrasting microscopic structures and intra-species variance:
\begin{itemize}
\item \textbf{Uniform Taxa}: Taxa with homogeneous growth rings (e.g., \textit{Guibourtia coleosperma}) are cleanly partitioned by linear centroid distance metrics (\textsf{Fixed Mahalanobis}).
\item \textbf{Outlier-Prone Taxa}: Taxa with severe heartwood/sapwood color transitions or surface abrasions (e.g., \textit{Afzelia bella}) distort distance metrics, requiring \textsf{Adversarial Validation} or \textsf{DataSAIL ILP} to isolate anomalous specimens.
\item \textbf{High Visual Similarity Taxa}: Taxa belonging to the same genus (e.g., \textit{Dalbergia oliveri} vs. \textit{Dalbergia tonkinensis}) require \textsf{Cosine Graph} component isolation to sever indirect visual leakage.
\end{itemize}

Enforcing a single splitting algorithm globally across all 18 species violates the \textbf{No Free Lunch theorem}. Allowing each taxon $c \in \{1, \dots, C\}$ to select its optimal solver $m_c \in \{1, \dots, K\}$ ($K=11$) yields a combinatorial configuration space of:
\begin{equation}
\mathcal{M} = K^C = 11^{18} \approx 5.5599 \times 10^{18} \quad \text{candidate partitions.}
\end{equation}

\subsubsection{Global Multi-Objective Fitness Function}
To evaluate candidate partitions on the assembled global dataset $\mathcal{D}(\boldsymbol{m}) = \bigcup_{c=1}^C \mathcal{D}_c(m_c)$, we define a Global Multi-Objective Fitness Function:
\begin{equation}
\mathrm{Fitness}(\boldsymbol{m}) = w_1 \cdot \left( \frac{L_{\text{DataSAIL}}(\boldsymbol{m})}{1000} \right) - w_2 \cdot \left( 10 \cdot MMD(\boldsymbol{m}) \right) - w_3 \cdot \left( 10 \cdot \mathrm{F1}_{\text{Hardest}}(\boldsymbol{m}) \right),
\label{eq:multi_obj_fitness}
\end{equation}
where $w_1 = 1.0$, $w_2 = 0.5$, and $w_3 = 0.5$. Minimizing $\mathrm{Fitness}(\boldsymbol{m})$ simultaneously:
\begin{enumerate}[label=\textbf{(\arabic*)}, leftmargin=*]
\item Minimizes DataSAIL inter-split leakage loss $L_{\text{DataSAIL}}(\boldsymbol{m})$.
\item Maximizes feature distribution discrepancy $MMD(\boldsymbol{m})$ (OOD evaluation challenge).
\item Maximizes hardest-class classification accuracy $\mathrm{F1}_{\text{Hardest}}(\boldsymbol{m})$ (inter-class decision boundary preservation).
\end{enumerate}

\subsubsection{Simulated Annealing Optimization Engine}
We solve this combinatorial optimization problem using **Simulated Annealing (SA)**:

\begin{enumerate}[label=\textbf{Step \arabic*:}, leftmargin=*]
\item \textbf{Pre-computation}: Candidate splits $\mathcal{D}_c(m_c)$ for all 18 taxa across all 11 solvers are pre-computed and cached.
\item \textbf{Initialization}: An initial configuration $\boldsymbol{m}^{(0)} = (m_1^{(0)}, \dots, m_{18}^{(0)})$ is selected randomly. The initial temperature is set to $T_0 = 10.0$ with a cooling rate $\alpha = 0.97$.
\item \textbf{Neighbor Mutation}: At step $t$, a random taxon $c$ is selected, and its solver is mutated to a random alternative $m_c' \in \{1, \dots, K\}$.
\item \textbf{Global Evaluation}: The assembled global dataset $\mathcal{D}(\boldsymbol{m}_{\text{cand}})$ is constructed, and $\Delta \mathrm{Fitness} = \mathrm{Fitness}(\boldsymbol{m}_{\text{cand}}) - \mathrm{Fitness}(\boldsymbol{m}^{(t)})$ is evaluated.
\item \textbf{Boltzmann Acceptance}: The candidate configuration is accepted with probability:
\begin{equation}
P(\text{accept}) = \begin{cases} 1.0 & \text{if } \Delta \mathrm{Fitness} < 0, \\ \exp\left(-\frac{\Delta \mathrm{Fitness}}{T_t}\right) & \text{if } \Delta \mathrm{Fitness} \ge 0. \end{cases}
\end{equation}
\item \textbf{Cooling}: Temperature is updated via $T_{t+1} = T_t \times 0.97$. The algorithm terminates after 400 iterations ($\approx 3$ seconds), returning the Pareto-optimal configuration $\boldsymbol{m}^*$.
\end{enumerate}
